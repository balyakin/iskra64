cmake_minimum_required(VERSION 3.16)
project(iskra64 VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(ISKRA64_STANDALONE ON)
else()
  set(ISKRA64_STANDALONE OFF)
endif()
option(ISKRA64_BUILD_TESTS "Build correctness and quality smoke tests" ${ISKRA64_STANDALONE})
option(ISKRA64_BUILD_BENCH "Build benchmarks" ${ISKRA64_STANDALONE})
option(ISKRA64_BENCH_NATIVE "Compile benchmarks for the build CPU (not portable)" ON)
option(ISKRA64_DISABLE_SIMD "Only compile the portable kernel" OFF)
option(ISKRA64_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(iskra64 STATIC src/iskra64.cpp)
add_library(iskra64::iskra64 ALIAS iskra64)
target_include_directories(iskra64 PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(iskra64 PUBLIC cxx_std_17)
if(ISKRA64_DISABLE_SIMD)
  target_compile_definitions(iskra64 PRIVATE ISKRA64_DISABLE_SIMD=1)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(iskra64 PRIVATE -Wall -Wextra -Wpedantic)
  if(ISKRA64_SANITIZERS)
    target_compile_options(iskra64 PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(iskra64 PUBLIC -fsanitize=address,undefined)
  endif()
endif()
if(ISKRA64_BUILD_TESTS)
  enable_testing()
  add_executable(iskra64_test tests/test.cpp)
  target_link_libraries(iskra64_test PRIVATE iskra64)
  add_test(NAME correctness COMMAND iskra64_test)
  add_executable(iskra64_quality tests/quality.cpp)
  target_link_libraries(iskra64_quality PRIVATE iskra64)
  add_test(NAME quality_smoke COMMAND iskra64_quality)
endif()
if(ISKRA64_BUILD_BENCH)
  add_executable(iskra64_bench bench/bench.cpp)
  target_link_libraries(iskra64_bench PRIVATE iskra64)
  target_include_directories(iskra64_bench PRIVATE third_party)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND ISKRA64_BENCH_NATIVE)
    target_compile_options(iskra64_bench PRIVATE -march=native)
  endif()
endif()
if(ISKRA64_STANDALONE)
  add_executable(iskra64_example examples/basic.cpp)
  target_link_libraries(iskra64_example PRIVATE iskra64)
endif()
if(ISKRA64_STANDALONE)
  install(TARGETS iskra64 EXPORT iskra64Targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  install(EXPORT iskra64Targets NAMESPACE iskra64:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iskra64)
  include(CMakePackageConfigHelpers)
  configure_package_config_file(cmake/iskra64Config.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/iskra64Config.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iskra64)
  write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/iskra64ConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/iskra64Config.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/iskra64ConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/iskra64)
endif()
